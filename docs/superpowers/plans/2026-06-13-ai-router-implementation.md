# Tianji AI Router Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the first version of Tianji AI Router: protocol-specific ordered failover chains over existing AI Gateways, with lightweight router logs and a lane-based route editor.

**Architecture:** AI Gateway remains the upstream credential, logging, pricing, and quota unit. AI Router adds a workspace resource, ordered router nodes, lightweight orchestration logs, and Express endpoints that use a shared gateway attempt executor. The first version routes only within compatible protocol lanes and never translates requests across protocol families.

**Tech Stack:** Prisma/PostgreSQL, Express, tRPC, Zod, React, TanStack Router, React Hook Form, Vitest, existing Tianji UI components.

```ts
export const AI_ROUTER_PROTOCOLS = {
  OPENAI_CHAT: 'openai-chat',
  OPENAI_RESPONSES: 'openai-responses',
  ANTHROPIC_MESSAGES: 'anthropic-messages',
} as const;

export type AIRouterProtocol =
  (typeof AI_ROUTER_PROTOCOLS)[keyof typeof AI_ROUTER_PROTOCOLS];

const chatProviders = new Set([
  'openai',
  'deepseek',
  'anthropic',
  'openrouter',
  'custom',
]);

const responseProviders = new Set(['openai', 'custom']);
const messageProviders = new Set(['anthropic', 'openrouter', 'custom']);

export function getAIRouterProtocolForPath(
  provider: string,
  endpoint: string
): AIRouterProtocol | null {
  if (endpoint === 'chat/completions' && chatProviders.has(provider)) {
    return AI_ROUTER_PROTOCOLS.OPENAI_CHAT;
  }

  if (endpoint === 'responses' && responseProviders.has(provider)) {
    return AI_ROUTER_PROTOCOLS.OPENAI_RESPONSES;
  }

  if (endpoint === 'messages' && messageProviders.has(provider)) {
    return AI_ROUTER_PROTOCOLS.ANTHROPIC_MESSAGES;
  }

  return null;
}

export function applyAIRouterModelOverride<T extends { model?: unknown }>(
  payload: T,
  modelOverride?: string | null
): T {
  if (!modelOverride) {
    return { ...payload };
  }

  return {
    ...payload,
    model: modelOverride,
  };
}

interface AIRouterGatewayEligibility {
  id: string;
  modelApiKey?: string | null;
  modelProvider?: string | null;
  modelProtocols?: string[];
}

interface AIRouterNodeEligibility {
  id: string;
  enabled: boolean;
  protocol: string;
  order: number;
  gateway: AIRouterGatewayEligibility | null;
}

export function selectEligibleAIRouterNodes<T extends AIRouterNodeEligibility>(
  nodes: T[],
  protocol: string
): T[] {
  return nodes
    .filter((node) => {
      const gateway = node.gateway;

      return (
        node.enabled &&
        node.protocol === protocol &&
        Boolean(gateway?.modelApiKey) &&
        Boolean(gateway?.modelProvider) &&
        Boolean(gateway?.modelProtocols?.includes(protocol))
      );
    })
    .sort((a, b) => a.order - b.order);
}

export interface AIRouterFailure {
  statusCode?: number;
  errorType?: 'network' | 'timeout' | 'unknown';
  retryableStatusCodes?: number[];
}

const DEFAULT_RETRYABLE_STATUS_CODES = new Set([429, 500, 502, 503, 504]);

export function isAIRouterRetryableFailure({
  statusCode,
  errorType,
  retryableStatusCodes,
}: AIRouterFailure): boolean {
  if (errorType === 'network' || errorType === 'timeout') {
    return true;
  }

  if (typeof statusCode !== 'number') {
    return false;
  }

  if (retryableStatusCodes && retryableStatusCodes.length > 0) {
    return retryableStatusCodes.includes(statusCode);
  }

  return DEFAULT_RETRYABLE_STATUS_CODES.has(statusCode);
}
```

```ts
export function validateAIGatewayRouterEligibilityInput(input: {
  modelApiKey?: string | null;
  modelProvider?: string | null;
  modelProtocols?: string[] | null;
}) {
  const hasModelApiKey = Boolean(input.modelApiKey);
  const modelProvider = input.modelProvider || null;
  const modelProtocols = input.modelProtocols ?? [];

  if (!hasModelApiKey) {
    return {
      modelProvider: modelProvider || null,
      modelProtocols,
    };
  }

  if (!modelProvider) {
    throw new Error('Provider is required when Model API Key is set');
  }

  if (modelProtocols.length === 0) {
    throw new Error(
      'At least one supported protocol is required when Model API Key is set'
    );
  }

  return {
    modelProvider,
    modelProtocols,
  };
}
```

```ts
async function runAIRouterAttempts(args: {
  workspaceId: string;
  routerId: string;
  protocol: AIRouterProtocol;
  executeAttempt: (node: any) => Promise<{
    ok: boolean;
    committed: boolean;
    gatewayId: string;
    logId?: string;
    statusCode?: number;
    failure?: { message: string; errorType?: string };
  }>;
}) {
  const start = Date.now();
  const router = await prisma.aIRouter.findFirst({
    where: {
      id: args.routerId,
      workspaceId: args.workspaceId,
      enabled: true,
    },
    include: {
      nodes: {
        where: { protocol: args.protocol },
        include: { gateway: true },
        orderBy: { order: 'asc' },
      },
    },
  });

  if (!router) {
    throw new Error('AI Router not found or disabled');
  }

  const nodes = selectEligibleAIRouterNodes(router.nodes, args.protocol);
  const attempts: AIRouterAttemptSummary[] = [];

  for (const node of nodes) {
    const result = await args.executeAttempt(node);
    const retryable = result.ok
      ? false
      : isAIRouterRetryableFailure({
          statusCode: result.statusCode,
          errorType:
            result.failure?.errorType === 'timeout' ||
            result.failure?.errorType === 'network'
              ? result.failure.errorType
              : 'unknown',
          retryableStatusCodes: node.retryableStatusCodes,
        });

    attempts.push({
      gatewayId: result.gatewayId,
      gatewayLogId: result.logId,
      statusCode: result.statusCode,
      retryable,
      errorType: result.failure?.errorType,
      message: result.failure?.message,
    });

    if (result.ok || result.committed || !retryable) {
      const log = await prisma.aIRouterLogs.create({
        data: {
          workspaceId: args.workspaceId,
          routerId: args.routerId,
          protocol: args.protocol,
          status: result.ok
            ? AIRouterLogsStatus.Success
            : result.committed
              ? AIRouterLogsStatus.Partial
              : AIRouterLogsStatus.Failed,
          finalGatewayId: result.gatewayId,
          finalGatewayLogId: result.logId,
          attemptGatewayIds: attempts.map((item) => item.gatewayId),
          attemptGatewayLogIds: attempts.flatMap((item) =>
            item.gatewayLogId ? [item.gatewayLogId] : []
          ),
          attemptErrors: serializeAIRouterAttemptErrors(attempts),
          attemptCount: attempts.length,
          duration: Date.now() - start,
        },
      });

      return { result, log };
    }
  }

  const last = attempts[attempts.length - 1];
  const log = await prisma.aIRouterLogs.create({
    data: {
      workspaceId: args.workspaceId,
      routerId: args.routerId,
      protocol: args.protocol,
      status: AIRouterLogsStatus.Failed,
      finalGatewayId: last?.gatewayId,
      finalGatewayLogId: last?.gatewayLogId,
      attemptGatewayIds: attempts.map((item) => item.gatewayId),
      attemptGatewayLogIds: attempts.flatMap((item) =>
        item.gatewayLogId ? [item.gatewayLogId] : []
      ),
      attemptErrors: serializeAIRouterAttemptErrors(attempts),
      attemptCount: attempts.length,
      duration: Date.now() - start,
    },
  });

  return { result: null, log };
}
```
